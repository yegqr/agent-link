for pj in node_modules/*/package.json node_modules/@*/*/package.json; do
  [ -f "$pj" ] || continue
  s=$(python3 -c "import json,sys; d=json.load(open(sys.argv[1])); s=d.get(\"scripts\",{}); print(\",\".join(f\"{k}={v}\" for k,v in s.items() if k in (\"preinstall\",\"install\",\"postinstall\",\"prepare\",\"prepublish\")))" "$pj" 2>/dev/null)
  [ -n "$s" ] && echo "$pj: $s"
done
